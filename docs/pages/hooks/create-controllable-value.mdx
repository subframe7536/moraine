`createControllableValue` bridges controlled and uncontrolled state for a custom component. It follows a caller-managed value when `value()` is defined, and otherwise manages its own state seeded from `defaultValue()`.

## Import

```tsx
import { createControllableValue } from 'moraine/utils'
```

## Signature

```ts
function createControllableValue<T>(
  options: CreateControllableValueOptions<T>,
): readonly [Accessor<T>, (update: T | ((previous: T) => T)) => void]

interface CreateControllableValueOptions<T> {
  value: Accessor<T | undefined>
  defaultValue: Accessor<T>
  onChange?: (value: T) => void
}
```

## Usage Example

```tsx title="src/CustomToggle.tsx"
import { createControllableValue } from 'moraine/utils'

interface CustomToggleProps {
  value?: boolean
  defaultValue?: boolean
  onChange?: (next: boolean) => void
}

export function CustomToggle(props: CustomToggleProps) {
  const [checked, setChecked] = createControllableValue({
    value: () => props.value,
    defaultValue: () => props.defaultValue ?? false,
  })

  function toggle() {
    const next = !checked()
    setChecked(next)
    props.onChange?.(next)
  }

  return (
    <button
      type="button"
      aria-pressed={checked()}
      onClick={toggle}
      class="px-3 py-1.5 rounded-md border"
    >
      {checked() ? 'Active' : 'Inactive'}
    </button>
  )
}
```

## Behavior

- **Controlled**: When `value()` returns a defined value, the accessor returns it. `setValue` does not change internal state; it only calls `onChange` with the next value, so the caller updates its own `value`.
- **Uncontrolled**: When `value()` returns `undefined`, the hook holds the state internally, seeded from `defaultValue()`. `setValue` updates that state and calls `onChange`.
- `setValue` skips `onChange` when the next value is `Object.is`-equal to the current one.
