`createDisclosureState` tracks open state, measures content height, and exposes data-attribute getters for expandable components such as `Collapsible` and `Accordion`.

## Import

```tsx
import { createDisclosureState } from 'moraine/utils'
```

## Signature

```ts
function createDisclosureState(options: CreateDisclosureStateOptions): {
  readonly contentHeight: Accessor<number>
  readonly dataAttrs: {
    readonly 'data-closed'?: string
    readonly 'data-disabled'?: string
    readonly 'data-expanded'?: string
  }
  readonly disabled: Accessor<boolean>
  /** Keeps the content shell mounted through its exit animation; returns a release function. */
  readonly registerContentElement: (element: HTMLElement) => () => void
  /** Measures the element's `scrollHeight` into `contentHeight`; returns a release function. */
  readonly registerMeasureElement: (element: HTMLElement) => () => void
}

interface CreateDisclosureStateOptions {
  disabled?: Accessor<boolean>
  open: Accessor<boolean>
}
```

## Usage Example

Register elements from the component that renders them, not inside the `ref` callback, so the
registration is released when the element unmounts.

```tsx title="src/MyCollapsible.tsx"
import { createEffect, createSignal, on, onCleanup } from 'solid-js'
import { createDisclosureState } from 'moraine/utils'

export function MyCollapsible() {
  const [open, setOpen] = createSignal(false)
  const disclosure = createDisclosureState({ open })
  const [panel, setPanel] = createSignal<HTMLDivElement>()

  createEffect(
    on(panel, (element) => {
      if (element) {
        onCleanup(disclosure.registerMeasureElement(element))
      }
    }),
  )

  return (
    <div>
      <button onClick={() => setOpen(!open())}>Toggle Panel</button>

      <div
        {...disclosure.dataAttrs}
        ref={setPanel}
        style={{
          '--mo-collapsible-content-height': `${disclosure.contentHeight()}px`,
        }}
        class="overflow-hidden transition-[height] duration-200"
      >
        <div class="p-4 bg-muted rounded-md">
          Panel content. Its height feeds the CSS variable above.
        </div>
      </div>
    </div>
  )
}
```

## Features

- Attaches a `ResizeObserver` to the content element and reports its `scrollHeight` through `contentHeight()`, remeasuring when the content changes size.
- Exposes `data-expanded`, `data-closed`, and `data-disabled` as getters, so each attribute tracks its own condition when you spread `dataAttrs`.
