## Import

Install `solid-toaster`:

<CodeTabs package="solid-toaster" />

Then import styles, mount a `Toaster` instance.

```tsx
import 'solid-toaster/style.css'

import { Button } from '@subframe7536/moraine'
import { Toaster, toast } from 'solid-toaster'

export default function App() {
  return (
    <>
      <Button onClick={() => toast.success('Saved!')}>Toast</Button>
      <Toaster />
    </>
  )
}
```

## Playground

<Playground
  controls={[
    {
      kind: 'input',
      prop: 'message',
      label: 'Message',
      defaultValue: 'Changes saved successfully',
    },
    {
      kind: 'select',
      prop: 'tone',
      label: 'Tone',
      defaultValue: 'success',
      options: [
        { label: 'Success', value: 'success' },
        { label: 'Warning', value: 'warning' },
        { label: 'Error', value: 'error' },
        { label: 'Info', value: 'info' },
      ],
    },
  ]}
>
  {(props) => {
    const showToast = () => {
      const message = props.message
      switch (props.tone) {
        case 'warning':
          toast.warning(message)
          break
        case 'error':
          toast.error(message)
          break
        case 'info':
          toast(message)
          break
        default:
          toast.success(message)
      }
    }

    return (
      <Button onClick={showToast} variant="outline" leading="i-lucide:bell">
        Trigger Toast Notification
      </Button>
    )

}}

</Playground>
