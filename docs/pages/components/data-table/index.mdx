<Playground
  controls={[
    {
      kind: 'select',
      prop: 'size',
      label: 'Size',
      defaultValue: 'md',
      options: [

        { label: 'Medium', value: 'md' },

      options: [
        { label: 'Row', value: 'row' },
        { label: 'Column', value: 'col' },
        { label: 'Both', value: 'both' },
      ],
    },
    { kind: 'switch', prop: 'sorting', label: 'Sorting', defaultValue: true },
    { kind: 'switch', prop: 'selectable', label: 'Selectable', defaultValue: true },
    { kind: 'switch', prop: 'pagination', label: 'Pagination', defaultValue: true },
    { kind: 'switch', prop: 'loading', label: 'Loading', defaultValue: false },
    { kind: 'switch', prop: 'remote', label: 'Remote Mode', defaultValue: false },
  ]}
>
  {(props) => {
    const [nameQuery, setNameQuery] = createSignal('')
    const [roleFilter, setRoleFilter] = createSignal(undefined)
    const [remoteSorting, setRemoteSorting] = createSignal([], {
      equals: (a, b) =>
        a.length === b.length && a.every((s, i) => s.id === b[i]?.id && s.desc === b[i]?.desc),
    })
    const [remotePagination, setRemotePagination] = createSignal(
      { pageIndex: 0, pageSize: 5 },
      {
        equals: (a, b) => a.pageIndex === b.pageIndex && a.pageSize === b.pageSize,
      },
    )
    const [remoteLoading, setRemoteLoading] = createSignal(false)
    const [serverRows, setServerRows] = createSignal(ALL_EMPLOYEES.slice(0, 5))
    const [serverTotal, setServerTotal] = createSignal(ALL_EMPLOYEES.length)

    const filteredLocalData = createMemo(() => {
      const q = nameQuery().trim().toLowerCase()
      const role = roleFilter()
      return ALL_EMPLOYEES.filter((emp) => {
        if (q && !emp.name.toLowerCase().includes(q)) return false
        if (role && role !== '' && emp.role !== role) return false
        return true
      })
    })

    createEffect(
      on(
        () => [props.remote, nameQuery(), roleFilter(), remoteSorting(), remotePagination()],
        () => {
          if (!props.remote) {
            setRemoteLoading(false)
            return
          }
          setRemoteLoading(true)
          const timer = setTimeout(() => {
            const q = nameQuery().trim().toLowerCase()
            const role = roleFilter()
            let filtered = ALL_EMPLOYEES.filter((emp) => {
              if (q && !emp.name.toLowerCase().includes(q)) return false
              if (role && role !== '' && emp.role !== role) return false
              return true
            })

            const sorting = remoteSorting()
            if (sorting.length > 0) {
              const sort = sorting[0]
              const key = sort.id
              filtered = [...filtered].sort((a, b) => {
                const va = String(a[key] ?? '')
                const vb = String(b[key] ?? '')
                const cmp = va.localeCompare(vb)
                return sort.desc ? -cmp : cmp
              })
            }

            const total = filtered.length
            setServerTotal(total)

            const { pageIndex, pageSize } = remotePagination()
            const start = pageIndex * pageSize
            const sliced = filtered.slice(start, start + pageSize)
            setServerRows(sliced)
            setRemoteLoading(false)
          }, 300)

          onCleanup(() => clearTimeout(timer))
        },
      ),
    )

    const hasFilter = () => Boolean(nameQuery()) || Boolean(roleFilter() && roleFilter() !== '')

    const handleReset = () => {
      setNameQuery('')
      setRoleFilter(undefined)
      if (props.remote) {
        setRemotePagination((prev) => ({ ...prev, pageIndex: 0 }))
      }
    }

    return (
      <div class="flex flex-col gap-3 w-full">
        <div class="flex flex-wrap items-center gap-2">
          <Input
            placeholder="Filter by name..."
            value={nameQuery()}
            onValueChange={(val) => {
              setNameQuery(String(val ?? ''))
              if (props.remote) {
                setRemotePagination((prev) => ({ ...prev, pageIndex: 0 }))
              }
            }}
            class="max-w-xs"
          />
          <Select
            placeholder="Filter role"
            items={ROLE_OPTIONS}
            value={roleFilter()}
            onValueChange={(val) => {
              setRoleFilter(val)
              if (props.remote) {
                setRemotePagination((prev) => ({ ...prev, pageIndex: 0 }))
              }
            }}
            allowClear
            class="w-44"
          />
          <Button
            variant="outline"
            onClick={handleReset}
            disabled={!hasFilter()}
          >
            Reset
          </Button>
        </div>

        <DataTable
          size={props.size}
          striped={props.striped}
          hoverEffect={props.hoverEffect}
          showBorder={props.showBorder}
          sorting={props.sorting}
          selectable={props.selectable}
          pagination={props.pagination}
          loading={props.loading || (props.remote && remoteLoading())}
          data={props.remote ? serverRows : filteredLocalData}
          columns={COLUMNS}
          rowCount={props.remote ? serverTotal() : undefined}
          manualPagination={props.remote}
          manualSorting={props.remote}
          currentPagination={props.remote ? remotePagination() : undefined}
          onPaginationChange={props.remote ? setRemotePagination : undefined}
          currentSorting={props.remote ? remoteSorting() : undefined}
          onSortingChange={props.remote ? setRemoteSorting : undefined}
          pageSizeOptions={[5, 10, 20]}
        />
      </div>
    )

}}

## Usage

Use `DataTable` when rendering structured, data-driven collections powered by TanStack Table.
It supports convenience mode (`data` and `columns`) for rapid development and instance mode (`table`)
for power-user headless configurations.

```tsx
import type { ColumnDef } from '@tanstack/solid-table'
import { DataTable } from 'moraine'

const data = [
  { id: '1', name: 'Alice', role: 'Engineer' },
  { id: '2', name: 'Bob', role: 'Designer' },
]

const columns: ColumnDef<any, (typeof data)[number]>[] = [
  { accessorKey: 'name', header: 'Name' },
  { accessorKey: 'role', header: 'Role' },
]

export function Example() {
  return <DataTable data={data} columns={columns} sorting selectable />
}
```

### Virtualization

Pass `virtualRender={virtualizer.virtualRender}` using `createTableVirtualizer` from `moraine/virtualizer` to virtualize large datasets while keeping native table semantics and column alignment.
