        { label: 'Error', value: 'error' },
        { label: 'Info', value: 'info' },
      ],
    },
    { kind: 'switch', prop: 'preventDuplicate', label: 'Prevent duplicate toasts', defaultValue: false },
    { kind: 'switch', prop: 'closeButton', label: 'Close button', defaultValue: true },
    { kind: 'switch', prop: 'action', label: 'Action button', defaultValue: true },
    {
      kind: 'input',
      prop: 'cancelText',
      label: 'Cancel text',
      defaultValue: 'Dismiss',
    },
    { kind: 'switch', prop: 'showProgress', label: 'Show progress bar', defaultValue: false },
  ]}
>
  {(props) => {
    const showToast = () => {
      const message = props.message
      const options = {
        description: props.description || undefined,
        preventDuplicate: props.preventDuplicate,
        closeButton: props.closeButton,
        showProgress: props.showProgress,
        action: props.action
          ? {
              label: 'Undo',
              onClick: () => {},
            }
          : undefined,
        cancel: props.cancelText
          ? {
              label: props.cancelText,
              onClick: () => {},
            }
          : undefined,
      }
      switch (props.tone) {
        case 'warning':
          Toast.warning(message, options)
          break
        case 'error':
          Toast.error(message, options)
          break
        case 'info':
          Toast.info(message, options)
          break
        default:
          Toast.success(message, options)
      }
    }

    return (
      <div class="p-6 flex items-center justify-center">
        <Button onClick={showToast} variant="outline" leading="i-lucide:bell">
          Trigger Toast Notification
        </Button>
      </div>
    )

}}

## Usage

Mount `<Toast />` once at your application root or inside a layout, then trigger toasts anywhere
in your app using static methods on `Toast`.

```tsx
import { Button, Toast } from 'moraine'

export default function App() {
  return (
    <>
      <Button onClick={() => Toast.success('Changes saved!')}>Save</Button>
      <Toast />
    </>
  )
}
```

### Semantic methods

Call dedicated tone methods for direct feedback, or use `Toast.show` for the default tone:

```tsx
Toast.show('Default message')
Toast.success('File uploaded successfully')
Toast.error('Network request failed')
Toast.warning('Storage is almost full')
Toast.info('New release available')
```

### Action and cancel buttons

Pass `action` or `cancel` configurations to prompt for immediate user interaction:

```tsx
Toast.show('Message archived', {
  action: {
    label: 'Undo',
    onClick: () => restoreMessage(),
  },
})
```

### Promises

Bind asynchronous tasks to pending, success, and error outcomes with `Toast.promise`:

```tsx
Toast.promise(saveSettings(), {
  loading: 'Saving profile...',
  success: 'Profile updated successfully',
  error: (err) => `Failed to save: ${String(err)}`,
})
```

### Custom rendering

Render arbitrary Solid JSX using `Toast.custom`:

```tsx
Toast.custom((id) => (
  <div class="p-4 bg-background border rounded-lg shadow-overlay flex items-center gap-3">
    <span>Custom banner notification</span>
    <Button size="sm" variant="ghost" onClick={() => Toast.dismiss(id)}>
      Dismiss
    </Button>
  </div>
))
```

### Accessibility and navigation

The toast container is rendered as an accessible notification region (`role="region"`) with a
persistent polite live region. Users can press <kbd>F6</kbd> anywhere on the page to focus the
notification list and navigate with <kbd>Tab</kbd>. Press <kbd>Escape</kbd> to collapse the
stack and restore focus. Dismiss individual toasts with the close button, action/cancel
controls, swipe gestures, or `Toast.dismiss`.
