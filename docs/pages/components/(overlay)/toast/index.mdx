import 'solid-toaster/style.css'
import { Button, Icon } from '@src'
import { toast, BaseToaster } from 'solid-toaster'

<Playground
  controls={[
    {
      kind: 'input',
      prop: 'message',
      label: 'Message',
      defaultValue: 'Changes saved successfully',
    },
    {
      kind: 'select',
      prop: 'tone',
      label: 'Tone',
      defaultValue: 'success',
      options: [
        { label: 'Success', value: 'success' },
        { label: 'Warning', value: 'warning' },
        { label: 'Error', value: 'error' },
        { label: 'Info', value: 'info' },
      ],
    },
    { kind: 'switch', prop: 'preventDuplicate', label: 'Prevent duplicate toasts', defaultValue: true },
  ]}
>
  {(props) => {
    const options = { toasterId: 'toast-playground' }
    const showToast = () => {
      const message = props.message
      switch (props.tone) {
        case 'warning':
          toast.warning(message, options)
          break
        case 'error':
          toast.error(message, options)
          break
        case 'info':
          toast.info(message, options)
          break
        default:
          toast.success(message, options)
      }
    }

    return (
      <>
        <BaseToaster
          id="toast-playground"
          preventDuplicate={props.preventDuplicate}
          icons={{
            success: () => <Icon name="icon-success" />,
            error: () => <Icon name="icon-error" />,
            warning: () => <Icon name="icon-warning" />,
            info: () => <Icon name="icon-info" />,
            close: () => <Icon name="icon-close" />,
          }}
          style={{
            '--normal-bg': 'var(--popover)',
            '--normal-text': 'var(--popover-foreground)',
            '--normal-border': 'var(--border)',
            '--border-radius': 'var(--radius)',
          }}
        />
        <Button onClick={showToast} variant="outline" leading="i-lucide:bell">
          Trigger Toast Notification
        </Button>
      </>
    )

}
}

</Playground>

## Usage

Use `solid-toaster` for temporary status that does not block the page. Moraine does not export a
Toast component. Use [Dialog](/components/dialog) when the user must confirm or complete a task.
Use [Tooltip](/components/tooltip) for a hint attached to a control. Mount the toaster host once
and use Moraine icons and theme variables to match your application.

```tsx
import 'solid-toaster/style.css'

import { Button, Icon } from 'moraine'
import { BaseToaster, toast } from 'solid-toaster'

export default function App() {
  return (
    <>
      <Button onClick={() => toast.success('Changes saved!')}>Save</Button>
      <BaseToaster
        visibleToasts={4}
        preventDuplicate
        style={{
          '--normal-bg': 'var(--popover)',
          '--normal-text': 'var(--popover-foreground)',
          '--normal-border': 'var(--border)',
          '--border-radius': 'var(--radius)',
        }}
        icons={{
          success: () => <Icon name="icon-success" />,
          error: () => <Icon name="icon-error" />,
          warning: () => <Icon name="icon-warning" />,
          info: () => <Icon name="icon-info" />,
          loading: () => <Icon name="icon-loading" class="animate-spin" />,
          close: () => <Icon name="icon-close" />,
        }}
      />
    </>
  )
}
```

### Install and mount the host

Install `solid-toaster`:

```shell title="bun" group-id="install-toaster"
bun add solid-toaster
```

```shell title="pnpm" group-id="install-toaster"
pnpm add solid-toaster
```

```shell title="npm" group-id="install-toaster"
npm i solid-toaster
```

Import `solid-toaster/style.css` and mount one `BaseToaster` at your application root before
calling `toast` methods. Its `icons` and `style` props connect the host to Moraine's semantic
icons and theme variables, as shown in the example above.

### Calls and lifecycles

Use semantic toast calls for immediate feedback and `toast.promise` for a promise whose pending,
success, and error states need distinct messages. Duration and scoped instances are supplied to the
toaster integration.

```tsx
toast.success('Changes saved successfully!')
toast.error('Failed to update repository settings.')
toast.promise(request, {
  loading: 'Sync in progress...',
  success: 'Sync finished',
  error: (error) => `Sync failed: ${String(error)}`,
})
```

### Duplicate messages

Set `preventDuplicate` on the host to avoid showing repeated toasts with the same content.

```tsx
<BaseToaster preventDuplicate />
```
