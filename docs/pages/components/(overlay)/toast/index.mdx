import { Button, Toaster, toast } from '@src'

<Playground
  controls={[
    {
      kind: 'input',
      prop: 'message',
      label: 'Message',
      defaultValue: 'Changes saved successfully',
    },
    {
      kind: 'select',
      prop: 'tone',
      label: 'Tone',
      defaultValue: 'success',
      options: [
        { label: 'Success', value: 'success' },
        { label: 'Warning', value: 'warning' },
        { label: 'Error', value: 'error' },
        { label: 'Info', value: 'info' },
      ],
    },
    { kind: 'switch', prop: 'preventDuplicate', label: 'Prevent duplicate toasts', defaultValue: true },
    { kind: 'switch', prop: 'closeButton', label: 'Close button', defaultValue: true },
    { kind: 'switch', prop: 'showProgress', label: 'Show progress bar', defaultValue: false },
  ]}
>
  {(props) => {
    const showToast = () => {
      const message = props.message
      switch (props.tone) {
        case 'warning':
          toast.warning(message)
          break
        case 'error':
          toast.error(message)
          break
        case 'info':
          toast.info(message)
          break
        default:
          toast.success(message)
      }
    }

    return (
      <div class="p-6 flex items-center justify-center">
        <Toaster
          preventDuplicate={props.preventDuplicate}
          closeButton={props.closeButton}
          showProgress={props.showProgress}
        />
        <Button onClick={showToast} variant="outline" leading="i-lucide:bell">
          Trigger Toast Notification
        </Button>
      </div>
    )

}}

</Playground>

## Usage

Mount `<Toaster />` once at your application root or inside a layout, then trigger toasts anywhere
in your app using the imperative `toast` manager.

```tsx
import { Button, Toaster, toast } from 'moraine'

export default function App() {
  return (
    <>
      <Button onClick={() => toast.success('Changes saved!')}>Save</Button>
      <Toaster />
    </>
  )
}
```

### Semantic methods

Call dedicated tone methods for direct feedback, or use the default `toast` function:

```tsx
toast('Default message')
toast.success('File uploaded successfully')
toast.error('Network request failed')
toast.warning('Storage is almost full')
toast.info('New release available')
```

### Action and cancel buttons

Pass `action` or `cancel` configurations to prompt for immediate user interaction:

```tsx
toast.add('Message archived', {
  action: {
    label: 'Undo',
    onClick: () => restoreMessage(),
  },
})
```

### Promises

Bind asynchronous tasks to pending, success, and error outcomes with `toast.promise`:

```tsx
toast.promise(saveSettings(), {
  loading: 'Saving profile...',
  success: 'Profile updated successfully',
  error: (err) => `Failed to save: ${String(err)}`,
})
```

### Custom rendering

Render arbitrary Solid JSX using `toast.custom`:

```tsx
toast.custom((id) => (
  <div class="p-4 bg-background border rounded-lg shadow-overlay flex items-center gap-3">
    <span>Custom banner notification</span>
    <Button size="sm" variant="ghost" onClick={() => toast.dismiss(id)}>
      Dismiss
    </Button>
  </div>
))
```

### Accessibility and navigation

The toast container is rendered as an accessible notification region (`role="region"`). Users can
press <kbd>F6</kbd> anywhere on the page to focus the notification list, navigate with <kbd>Tab</kbd>,
and dismiss individual toasts with <kbd>Escape</kbd> or swipe gestures.
